sqlplus / as sysdba <<'EOF'

-- Step 1: Zerar o AMM (NUNCA use SCOPE=BOTH aqui)
ALTER SYSTEM SET MEMORY_TARGET=0 SCOPE=SPFILE;
ALTER SYSTEM SET MEMORY_MAX_TARGET=0 SCOPE=SPFILE;

-- Step 2: Definir limites de SGA
ALTER SYSTEM SET SGA_MAX_SIZE=1500M SCOPE=SPFILE;
ALTER SYSTEM SET SGA_TARGET=1500M SCOPE=SPFILE;

-- Step 3: Definir limites de PGA
ALTER SYSTEM SET PGA_AGGREGATE_TARGET=500M SCOPE=SPFILE;
ALTER SYSTEM SET PGA_AGGREGATE_LIMIT=1000M SCOPE=SPFILE;

-- Step 4: Limitar processos (reduz concorrência = menos RAM)
ALTER SYSTEM SET PROCESSES=150 SCOPE=SPFILE;

-- Confirmação antes de rebootar
SHOW PARAMETER MEMORY_TARGET;
SHOW PARAMETER SGA_TARGET;
SHOW PARAMETER PGA_AGGREGATE_TARGET;

EXIT;
EOF
